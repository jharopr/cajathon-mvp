<script>
        (function () {
                var raiz = document.getElementById('snm-avisos');
                if (!raiz) {
                        return;
                }
                try {
                        if (sessionStorage.getItem('snm-avisos-cerrados') === '1') {
                                raiz.hidden = true;
                                return;
                        }
                } catch (e) {
                }
                var pista = raiz.querySelector('.snm-avisos__pista');
                var items = raiz.querySelectorAll('.snm-avisos__item');
                var puntos = raiz.querySelector('.snm-avisos__puntos');
                var total = items.length;
                var actual = 0;
                var temporizador = null;
                var reducido = window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches;

                if (total < 2) {
                        raiz.classList.add('snm-avisos--uno');
                }

                for (var i = 0; i < total; i++) {
                        var punto = document.createElement('button');
                        punto.type = 'button';
                        punto.className = 'snm-avisos__punto';
                        punto.setAttribute('aria-label', 'Aviso ' + (i + 1) + ' de ' + total);
                        punto.setAttribute('data-indice', i);
                        puntos.appendChild(punto);
                }

                function mostrar(indice) {
                        actual = (indice + total) % total;
                        pista.style.transform = 'translateX(' + (-100 * actual) + '%)';
                        Array.prototype.forEach.call(items, function (item, n) {
                                item.setAttribute('aria-hidden', n === actual ? 'false' : 'true');
                                Array.prototype.forEach.call(item.querySelectorAll('a'), function (a) {
                                        a.tabIndex = n === actual ? 0 : -1;
                                });
                        });
                        Array.prototype.forEach.call(puntos.children, function (p, n) {
                                p.setAttribute('aria-current', n === actual ? 'true' : 'false');
                        });
                }

                function detener() {
                        clearInterval(temporizador);
                        temporizador = null;
                }

                function iniciar() {
                        detener();
                        if (total > 1 && !reducido) {
                                temporizador = setInterval(function () {
                                        mostrar(actual + 1);
                                }, 3000);
                        }
                }

                raiz.addEventListener('click', function (e) {
                        var boton = e.target.closest('button');
                        if (!boton) {
                                return;
                        }
                        var accion = boton.getAttribute('data-avisos');
                        if (accion === 'anterior') {
                                mostrar(actual - 1);
                                iniciar();
                        } else if (accion === 'siguiente') {
                                mostrar(actual + 1);
                                iniciar();
                        } else if (accion === 'cerrar') {
                                detener();
                                raiz.hidden = true;
                                try {
                                        sessionStorage.setItem('snm-avisos-cerrados', '1');
                                } catch (err) {
                                }
                        } else if (boton.hasAttribute('data-indice')) {
                                mostrar(parseInt(boton.getAttribute('data-indice'), 10));
                                iniciar();
                        }
                });

                raiz.addEventListener('mouseenter', detener);
                raiz.addEventListener('mouseleave', iniciar);
                raiz.addEventListener('focusin', detener);
                raiz.addEventListener('focusout', iniciar);

                var inicioX = null;
                var ventana = raiz.querySelector('.snm-avisos__ventana');
                ventana.addEventListener('touchstart', function (e) {
                        inicioX = e.touches[0].clientX;
                        detener();
                }, {passive: true});
                ventana.addEventListener('touchend', function (e) {
                        if (inicioX === null) {
                                return;
                        }
                        var dx = e.changedTouches[0].clientX - inicioX;
                        if (Math.abs(dx) > 40) {
                                mostrar(actual + (dx < 0 ? 1 : -1));
                        }
                        inicioX = null;
                        iniciar();
                });

                mostrar(0);
                iniciar();
        })();
</script>
